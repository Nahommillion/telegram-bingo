function init() { // Always read the current Telegram Web App launch data. const webApp = window.Telegram?.WebApp; return webApp && typeof webApp.initData === 'string' ? webApp.initData : ''; } async function api(url, opt = {}) { const webApp = window.Telegram?.WebApp; const initData = webApp?.initData || ''; if (!initData) { throw new Error( 'Telegram launch data is missing. Close the Mini App, return to your bot chat, and tap Play Now again.' ); } const headers = new Headers(opt.headers || {}); headers.set('Content-Type', 'application/json'); headers.set('X-Telegram-Init-Data', initData); const response = await fetch(url, { ...opt, headers }); const contentType = response.headers.get('content-type') || ''; let data; if (contentType.includes('application/json')) { data = await response.json(); } else { const responseText = await response.text(); data = { detail: responseText || 'The server returned a non-JSON response.' }; } if (!response.ok) { throw new Error( data.detail || data.message || `Request failed (${response.status})` ); } return data; }
